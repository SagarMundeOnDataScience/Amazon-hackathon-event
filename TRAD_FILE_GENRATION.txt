# =============================================================================
#  STEP 13/14 -- TRAD FILE GENERATION
#
#  PySpark translation of AZBJ_VALUATION_FILE_GEN_PKG (June-2026 version):
#      PROCEDURE trad_file  -> generate_trad_file()   378 fields  VAL_TRAD_<m>.TXT
#      PROCEDURE spg_file   -> generate_spg_file()    274 fields  VAL_SPG_<m>.TXT
#  (trad_csc_file and ace_file are COMMENTED OUT in all_files -- not generated,
#   matching the June run. Products 275/291 and the group products therefore
#   reach NO file, exactly as in Oracle.)
#
#  Built on the proven conventions of Annuity_File_Gen.py / ul_file_gen.py
#  (file_generation_spec.md rules R1-R6): explicit coalesce-concat (never
#  concat_ws), byte-verbatim strings (never trim), oracle_num_to_str (never
#  cast-to-string), dd/MM/yyyy dates in an Asia/Kolkata session, one file via
#  staging + rename, structural preflight before writing.
#
#  TRAD-SPECIFIC FACTS, all read off the package source:
#   * Field order comes from the V_LINE CONCATENATION, not the cursor SELECT.
#     They differ once: the cursor selects ...basic_sa, branch_code,
#     os_adv_inst, date_last_adv_due... but v_line emits branch_code,
#     DATE_LAST_ADV_DUE, OS_ADV_INST (swapped). The lists below follow v_line.
#   * The last two emitted fields (COMBO_PRODUCT_CODE, SOLUTION_ID -- call
#     51247) are MISSING from the cursor SELECT in the source copy we were
#     given; the compiled production version must select them. Both are
#     NOT_PRODUCED (always NULL) -> two empty trailing fields.
#   * BONUS_RATE is computed AT FILE TIME: a scalar subquery on
#     AZBJ_BONUS_RATES (bonus_type='OTB', bonus_from_date=01-APR-2025,
#     product_id IN (329,359), cover_code IN ('RP10A01','L182A01','L182B01')),
#     gated on the ROW's com_date <= 31-Mar-2025. Date comparisons are done at
#     DATE level (the platform copy carries the +05:30 shift). A key with >1
#     rates rows would be ORA-01427 in Oracle (file aborts) -> we raise.
#   * CASE transforms: salaried_yn / prime_disc_flag YES->Y, NO->N, else
#     UPPER; income_frequency_type = 'AD' else NULL.
#   * Aliases: channel_code<-irda_code, age_proof<-st_nst_flag,
#     ib_amount<-annuity_amount, ib_frequency<-annuity_frequency.
#   * ROUND(x) is Oracle half-away-from-zero, 0 dp == Spark F.round(x, 0).
#   * JUNE EXCLUSIONS: the 12 registered ours-only contracts (4 D-TRAD-11 +
#     8 recovered pc) do not exist in Oracle's June table, so they must not
#     reach the June files. Flag-gated; OFF for any non-June run.
#   * No ORDER BY in the cursors (parallel 20): row order is not part of the
#     contract. Compare with verify_txt_vs_oracle.py (sorted).
# =============================================================================

# =============================================================================
# Value rendering (file_generation_spec.md R1-R3; identical to the proven
# Annuity/UL implementations)
# =============================================================================

def _scale_of(dtype):
    if dtype.startswith("decimal"):
        return int(dtype[dtype.index(",") + 1:dtype.index(")")])
    if dtype.startswith(("int", "bigint", "smallint", "tinyint")):
        return 0
    return 10


def oracle_num_to_str(col, scale):
    """Implicit NUMBER -> VARCHAR2: shortest form, never scientific.
    6065.0000000000->'6065', 0.5->'.5', -0.5->'-.5', 0->'0'."""
    s = F.format_number(col.cast(T.DecimalType(38, scale)), scale)
    s = F.regexp_replace(s, ",", "")
    if scale > 0:
        s = F.regexp_replace(s, r"(\.[0-9]*?)0+$", r"$1")
        s = F.regexp_replace(s, r"\.$", "")
        s = F.regexp_replace(s, r"^(-?)0\.", r"$1.")
    return s


def oracle_date_to_str(col):
    """TO_CHAR(d,'DD/MM/YYYY'); to_date first (+05:30 shift; IST session)."""
    return F.date_format(F.to_date(col), "dd/MM/yyyy")


def render_field(df, name, kind, src):
    """Render one field exactly as the package's cursor/v_line does."""
    dtypes = dict(df.dtypes)
    if src not in dtypes:
        raise ValueError(f"source column missing: {src} (field {name})")
    col = F.col("`" + src + "`")
    dtype = dtypes[src]

    if kind == "date":
        rendered = oracle_date_to_str(col)
    elif kind == "round":
        # ROUND(x): half away from zero, 0 dp; NULL stays NULL
        rendered = oracle_num_to_str(F.round(col.cast(T.DecimalType(38, 10)), 0), 0)
    elif kind == "case_yn":
        u = F.upper(col)
        rendered = (F.when(u == "YES", F.lit("Y"))
                     .when(u == "NO", F.lit("N"))
                     .otherwise(u))
    elif kind == "case_ad":
        rendered = F.when(col == "AD", F.lit("AD"))
    elif kind == "bonus_rate":
        # the joined _BONUS_RATE_CALC column (see _with_bonus_rate)
        c2 = F.col("_BONUS_RATE_CALC")
        rendered = oracle_num_to_str(c2, 10)
    elif dtype.startswith(("date", "timestamp")):
        # raw date field without TO_CHAR in the package -- only
        # BASE_LIFE_DESC_DATE (all NULL). Preflight warns if non-null.
        rendered = oracle_date_to_str(col)
    elif dtype.startswith(("decimal", "int", "bigint", "smallint",
                           "double", "float", "tinyint")):
        rendered = oracle_num_to_str(col, _scale_of(dtype))
    else:
        rendered = col.cast("string")      # strings byte-verbatim, NO TRIM

    return F.coalesce(rendered, F.lit(""))


def build_line(df, fields):
    """Explicit coalesce-concat. NEVER concat_ws (it drops NULLs AND their
    separators, shifting every later field)."""
    parts = []
    for i, (name, kind, src) in enumerate(fields):
        if i:
            parts.append(F.lit(","))
        parts.append(render_field(df, name, kind, src))
    return F.concat(*parts)


# =============================================================================
# BONUS_RATE -- the scalar subquery, as a join
# =============================================================================

def _with_bonus_rate(df, spark):
    if not BONUS_RATES_PATH:
        raise ValueError("BONUS_RATES_PATH not set -- the BONUS_RATE field "
                         "is computed from azbj_bonus_rates at file time.")
    br = spark.read.format("delta").load(BONUS_RATES_PATH)
    br = br.toDF(*[c.upper() for c in br.columns])
    dim = (br.filter((F.col("BONUS_TYPE") == "OTB")
                     & (F.to_date(F.col("BONUS_FROM_DATE"))
                        == F.lit("2025-04-01").cast("date"))
                     & F.col("PRODUCT_ID").isin(329, 359)
                     & F.col("COVER_CODE").isin("RP10A01", "L182A01",
                                                "L182B01"))
             .groupBy(F.col("PRODUCT_ID").alias("BR_PID"),
                      F.col("COVER_CODE").alias("BR_COV"))
             .agg(F.first("BONUS_RATE").alias("_BR_RATE"),
                  F.count(F.lit(1)).alias("_BR_N")))
    bad = dim.filter(F.col("_BR_N") > 1).count()
    if bad:
        raise ValueError(f"azbj_bonus_rates: {bad} (product,cover) key(s) "
                         "with >1 OTB rows -- Oracle's scalar subquery would "
                         "raise ORA-01427 and abort the file. Resolve first.")
    out = (df.join(dim, (df["PRODUCT_ID"] == F.col("BR_PID"))
                   & (df["COVER_CODE"] == F.col("BR_COV")), "left")
             .withColumn("_BONUS_RATE_CALC",
                         F.when(F.col("BR_PID").isNotNull()
                                & (F.to_date(F.col("COM_DATE"))
                                   <= F.lit("2025-03-31").cast("date")),
                                F.col("_BR_RATE")))
             .drop("BR_PID", "BR_COV", "_BR_RATE", "_BR_N"))
    return out

# Structural preflight (file_generation_spec.md section 3)
# =============================================================================

def _preflight(df, fields, label):
    dtypes = dict(df.dtypes)
    missing = [s for _, k, s in fields
               if k != "bonus_rate" and s not in dtypes]
    if missing:
        raise ValueError(f"{label}: source is missing {sorted(set(missing))}")

    checks = []
    for name, kind, src in fields:
        if kind == "bonus_rate":
            continue
        dt = dtypes[src]
        q = "`" + src + "`"
        if dt == "string" and kind == "raw":
            checks.append(F.sum(F.when(F.col(q).contains(","), 1)
                                 .otherwise(0)).alias("COMMA|" + src))
            checks.append(F.sum(F.when(F.col(q).rlike(r"[\r\n]"), 1)
                                 .otherwise(0)).alias("NEWLINE|" + src))
        if kind == "raw" and dt.startswith(("date", "timestamp")):
            checks.append(F.sum(F.when(F.col(q).isNotNull(), 1)
                                 .otherwise(0)).alias("RAWDATE|" + src))
        if dt.startswith(("decimal", "double", "float")):
            checks.append(F.sum(F.when(
                render_field(df, name, kind, src).rlike("[eE]"), 1)
                .otherwise(0)).alias("SCI|" + src))

    row = df.agg(*checks).collect()[0].asDict() if checks else {}
    problems = {k: v for k, v in row.items() if v}
    if problems:
        msg = (f"{label}: structural findings -> {problems}"
               "\n  COMMA|/NEWLINE| break the record layout (NOTE: Step-12"
               " stage 3 replaced commas in channel -- any remaining comma"
               " is a new source value; compare with Oracle before override)."
               "\n  RAWDATE| a date field the package emits WITHOUT TO_CHAR"
               " has non-null values -- Oracle would print NLS format; check."
               "\n  SCI| scientific notation reached the rendered output.")
        if STRICT:
            raise ValueError(msg)
        logger.warning(msg)


# -----------------------------------------------------------------------------
# WRITE STRATEGY (changed 2026-10-20 after repeated 'Broken pipe' failures).
#
# The original coalesce(1) write streams ONE ~7GB multipart upload through a
# single task for 30+ minutes; the platform's S3 gateway drops connections of
# that lifetime, and no retry setting fixes a pattern the gateway refuses.
#
# "s3_multipart_copy" (default) writes the SAME lines as ~ROWS_PER_PART-row
# parallel part files (each upload lasts a minute or two, task-level retries
# cheap), then assembles the single TXT SERVER-SIDE with S3 UploadPartCopy:
# the final object is stitched from the parts inside the storage system --
# no bytes stream through our client, each copy call is near-instant.
# BYTE-IDENTICAL output: every part ends with LINE_SEP and parts concatenate
# in part-number order; row order was never part of the contract (Oracle's
# cursor has no ORDER BY).
#
# Needs boto3 on the driver (import checked up front) and the S3 endpoint +
# keys from the BALIC_S3_* environment variables (never hardcoded), falling
# back to the session's fs.s3a.* Hadoop conf. S3 constraint honoured: every
# part except the last must be >= 5MB -- ROWS_PER_PART=120000 gives ~60-90MB
# parts; a violation raises with instructions instead of writing a bad file.
#
# WRITE_MODE = "coalesce" restores the original single-task behaviour.
# -----------------------------------------------------------------------------
import os

WRITE_MODE = "s3_multipart_copy"     # or "coalesce"
ROWS_PER_PART = 120000
MIN_PART_BYTES = 5 * 1024 * 1024     # S3 multipart minimum (all but last)


def _split_s3(uri):
    for scheme in ("s3a://", "s3://"):
        if uri.startswith(scheme):
            bucket, _, key = uri[len(scheme):].partition("/")
            return bucket, key
    raise ValueError(f"not an s3a path: {uri}")


def _boto_client(spark):
    try:
        import boto3
    except ImportError:
        raise RuntimeError(
            "boto3 is not available on the driver. Either install it, or set "
            "WRITE_MODE='coalesce' to use the original single-task write.")
    hconf = spark._jsc.hadoopConfiguration()
    endpoint = os.environ.get("BALIC_S3_ENDPOINT") \
        or hconf.get("fs.s3a.endpoint")
    access = os.environ.get("BALIC_S3_ACCESS_KEY") \
        or hconf.get("fs.s3a.access.key")
    secret = os.environ.get("BALIC_S3_SECRET_KEY") \
        or hconf.get("fs.s3a.secret.key")
    if not (endpoint and access and secret):
        raise RuntimeError("S3 endpoint/credentials not found in BALIC_S3_* "
                           "env vars or fs.s3a.* conf")
    if not endpoint.startswith("http"):
        scheme = "https" if hconf.get(
            "fs.s3a.connection.ssl.enabled", "true") == "true" else "http"
        endpoint = f"{scheme}://{endpoint}"
    return boto3.client("s3", endpoint_url=endpoint,
                        aws_access_key_id=access,
                        aws_secret_access_key=secret)


def _concat_on_s3(spark, staging, final):
    """Assemble `final` from the part files under `staging`, server-side."""
    s3 = _boto_client(spark)
    bucket, skey = _split_s3(staging)
    fbucket, fkey = _split_s3(final)
    if fbucket != bucket:
        raise ValueError("staging and final must be in the same bucket")

    parts, token = [], None
    while True:
        kw = dict(Bucket=bucket, Prefix=skey.rstrip("/") + "/")
        if token:
            kw["ContinuationToken"] = token
        resp = s3.list_objects_v2(**kw)
        for o in resp.get("Contents", []):
            name = o["Key"].rsplit("/", 1)[-1]
            if name.startswith("part-") and o["Size"] > 0:
                parts.append((o["Key"], o["Size"]))
        if not resp.get("IsTruncated"):
            break
        token = resp["NextContinuationToken"]
    if not parts:
        raise RuntimeError(f"no part files under {staging}")
    parts.sort(key=lambda p: p[0])      # part-00000, part-00001, ...

    small = [k for k, sz in parts[:-1] if sz < MIN_PART_BYTES]
    if small:
        raise RuntimeError(
            f"{len(small)} non-final part(s) below the 5MB S3 multipart "
            f"minimum (first: {small[0]}). Lower ROWS_PER_PART's divisor -- "
            "i.e. RAISE ROWS_PER_PART so parts get bigger -- and rerun.")

    logger.info("concat: %d parts, %s bytes total -> %s",
                len(parts), f"{sum(sz for _, sz in parts):,}", final)
    mpu = s3.create_multipart_upload(Bucket=bucket, Key=fkey)
    try:
        etags = []
        for i, (key, _sz) in enumerate(parts, 1):
            r = s3.upload_part_copy(
                Bucket=bucket, Key=fkey, UploadId=mpu["UploadId"],
                PartNumber=i, CopySource={"Bucket": bucket, "Key": key})
            etags.append({"PartNumber": i,
                          "ETag": r["CopyPartResult"]["ETag"]})
        s3.complete_multipart_upload(
            Bucket=bucket, Key=fkey, UploadId=mpu["UploadId"],
            MultipartUpload={"Parts": etags})
    except Exception:
        s3.abort_multipart_upload(Bucket=bucket, Key=fkey,
                                  UploadId=mpu["UploadId"])
        raise
    # staging cleanup (parts + markers)
    token = None
    while True:
        kw = dict(Bucket=bucket, Prefix=skey.rstrip("/") + "/")
        if token:
            kw["ContinuationToken"] = token
        resp = s3.list_objects_v2(**kw)
        keys = [{"Key": o["Key"]} for o in resp.get("Contents", [])]
        if keys:
            s3.delete_objects(Bucket=bucket, Delete={"Objects": keys})
        if not resp.get("IsTruncated"):
            break
        token = resp["NextContinuationToken"]


def _write_single_txt(spark, line_df, file_name, n_rows):
    staging = f"{FILE_DIR}/_staging_{file_name}"
    final = f"{FILE_DIR}/{file_name}"

    if WRITE_MODE == "s3_multipart_copy":
        nparts = max(1, (n_rows + ROWS_PER_PART - 1) // ROWS_PER_PART)
        logger.info("writing %s rows as %d part(s), then server-side concat",
                    f"{n_rows:,}", nparts)
        (line_df.repartition(nparts).write.mode("overwrite")
                .option("lineSep", LINE_SEP).text(staging))
        _concat_on_s3(spark, staging, final)
        logger.info("wrote %s", final)
        return final

    # original behaviour: single-task write + rename
    (line_df.coalesce(1).write.mode("overwrite")
            .option("lineSep", LINE_SEP).text(staging))
    jvm = spark._jvm
    hconf = spark._jsc.hadoopConfiguration()
    Path = jvm.org.apache.hadoop.fs.Path
    staging_p, final_p = Path(staging), Path(final)
    fs = staging_p.getFileSystem(hconf)
    parts = [st.getPath() for st in fs.listStatus(staging_p)
             if st.getPath().getName().startswith("part-")]
    if len(parts) != 1:
        raise RuntimeError(f"expected one part file, found {len(parts)}")
    if fs.exists(final_p):
        fs.delete(final_p, False)
    if not fs.rename(parts[0], final_p):
        raise RuntimeError(f"could not rename {parts[0]} -> {final_p}")
    fs.delete(staging_p, True)
    logger.info("wrote %s", final)
    return final

def _load_source(spark):
    df = spark.read.format("delta").load(SRC_PATH)
    df = df.toDF(*[c.upper() for c in df.columns])
    if APPLY_JUNE_EXCLUSIONS:
        n0 = df.count()
        df = df.filter(~F.col("CONTRACT_ID").cast("long")
                        .isin(JUNE_EXCLUDE_IDS))
        n1 = df.count()
        logger.info("June exclusions: %d rows removed (expect 23); "
                    "%d -> %d", n0 - n1, n0, n1)
    return df


def generate_trad_file(spark, src_df, p_month=P_MONTH):
    """trad_file -> VAL_TRAD_<p_month>.TXT, 378 fields,
    product_id NOT IN the group/CSC set."""
    df = _with_bonus_rate(src_df, spark)
    sel = df.filter(~F.col("PRODUCT_ID").cast("long")
                     .isin(TRAD_EXCLUDE_PRODUCTS))
    _preflight(sel, TRAD_FIELDS, "trad_file")
    n = sel.count()
    logger.info("trad_file: %s rows (product NOT IN %s)",
                f"{n:,}", TRAD_EXCLUDE_PRODUCTS)
    line_df = sel.select(build_line(sel, TRAD_FIELDS).alias("value"))
    path = _write_single_txt(spark, line_df, f"VAL_TRAD_{p_month}.TXT", n)
    logger.info("trad_file: %s records x %d fields -> %s",
                f"{n:,}", len(TRAD_FIELDS), path)
    return path, n


def generate_spg_file(spark, src_df, p_month=P_MONTH):
    """spg_file -> VAL_SPG_<p_month>.TXT, 274 fields, product_id = 321.
    321 rows ALSO appear in VAL_TRAD (it is not in the exclusion list) --
    that duplication is Oracle's, by design."""
    sel = src_df.filter(F.col("PRODUCT_ID").cast("long").isin(SPG_PRODUCT_IDS))
    _preflight(sel, SPG_FIELDS, "spg_file")
    n = sel.count()
    logger.info("spg_file: %s rows (product 321)", f"{n:,}")
    line_df = sel.select(build_line(sel, SPG_FIELDS).alias("value"))
    path = _write_single_txt(spark, line_df, f"VAL_SPG_{p_month}.TXT", n)
    logger.info("spg_file: %s records x %d fields -> %s",
                f"{n:,}", len(SPG_FIELDS), path)
    return path, n


def main():
    for nm, v in (("SRC_PATH", SRC_PATH), ("FILE_DIR", FILE_DIR),
                  ("BONUS_RATES_PATH", BONUS_RATES_PATH)):
        if not v:
            raise ValueError(f"{nm} not set")
    spark = (SparkSession.builder.appName("trad_file_gen")
             # MANDATORY: a UTC session renders the previous calendar day for
             # every +05:30-shifted date column.
             .config("spark.sql.session.timeZone", "Asia/Kolkata")
             .getOrCreate())
    spark.conf.set("spark.sql.session.timeZone", "Asia/Kolkata")
    assert spark.conf.get("spark.sql.session.timeZone") == "Asia/Kolkata"

    df = _load_source(spark)
    total = df.count()

    _, n_trad = generate_trad_file(spark, df)
    _, n_spg = generate_spg_file(spark, df)

    n_excluded_products = df.filter(
        F.col("PRODUCT_ID").cast("long").isin(TRAD_EXCLUDE_PRODUCTS)).count()
    logger.info("accounting: VAL_TRAD %s + excluded-product rows %s = %s ; "
                "table (after June exclusions) %s ; VAL_SPG %s (321 subset, "
                "also inside VAL_TRAD)",
                f"{n_trad:,}", f"{n_excluded_products:,}",
                f"{n_trad + n_excluded_products:,}", f"{total:,}", f"{n_spg:,}")
    if n_trad + n_excluded_products != total:
        raise ValueError("trad_file filter does not partition the table -- "
                         "rows dropped or duplicated")
    spark.stop()


if __name__ == "__main__":
    main()
